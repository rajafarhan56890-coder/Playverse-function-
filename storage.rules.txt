rules_version = '2';

service firebase.storage {
  match /b/{bucket}/o {

    function isSignedIn() {
      return request.auth != null;
    }

    // User avatars — user can upload/replace only their own, 5MB cap, images only.
    // This is the only Firebase Storage usage left in PlayVerse: game images,
    // offer images, and branding/banner assets are now Cloudinary-hosted URLs
    // entered directly in the admin panel and stored as plain string fields
    // in Firestore (games.imageURL, offers.imageURL, settings.logoURL) —
    // there is no Storage upload path for any of those anymore.
    match /avatars/{uid}/{fileName} {
      allow read: if true;
      allow write: if isSignedIn() && request.auth.uid == uid
        && request.resource.size < 5 * 1024 * 1024
        && request.resource.contentType.matches('image/.*');
    }

    // Default deny — anything not explicitly matched above is blocked,
    // including the old /games, /offers, and /branding paths.
    match /{allPaths=**} {
      allow read, write: if false;
    }
  }
}
